WITH ParesMarcaciones AS ( SELECT m1.usuario_id, m1.ticket_id, DATE(m1.fecha_hora AT TIME ZONE 'America/Lima') AS fecha_reporte, m1.fecha_hora AS hora_ingreso, MIN(m2.fecha_hora) AS hora_salida FROM marcaciones m1 LEFT JOIN marcaciones m2 ON m1.usuario_id = m2.usuario_id AND m1.ticket_id = m2.ticket_id AND m2.tipo = 'SALIDA' AND m2.fecha_hora > m1.fecha_hora WHERE m1.tipo = 'INGRESO' GROUP BY m1.usuario_id, m1.ticket_id, m1.fecha_hora ) SELECT u.nombre AS tecnico, t.numero_ticket, l.nombre AS local, p.fecha_reporte, TO_CHAR(p.hora_ingreso AT TIME ZONE 'America/Lima', 'HH24:MI:SS') AS ingreso, TO_CHAR(p.hora_salida AT TIME ZONE 'America/Lima', 'HH24:MI:SS') AS salida, ROUND(EXTRACT(EPOCH FROM (p.hora_salida - p.hora_ingreso)) / 3600.0, 2) AS horas_totales FROM ParesMarcaciones p JOIN usuarios u ON p.usuario_id = u.id JOIN tickets t ON p.ticket_id = t.id JOIN locales l ON t.local_id = l.id WHERE p.hora_salida IS NOT NULL AND EXTRACT(MONTH FROM p.fecha_reporte) = 3;
